+    // Diagnostic only: emulate Tokio's scalar stream short-read policy. An urgent-data boundary can leave unread ordinary bytes behind this cleared readiness.
+    pub(crate) async fn socket_monitor_recv_short_read_poll(
+        &self,
+        monitor: &mut Storage,
+        established: &mut bool,
+        buffer: &mut [u8],
+        cancellation: &Cancellation,
+    ) -> OperationResult<usize> {
+        let state = monitor.as_ref();
+        let registration = state.registration();
+        let mut route = None;
+        let mut wait = Wait::new(registration);
+        let mut cancelled = pin!(cancellation.cancelled());
+        let mut wait_first = *established;
+        poll_fn(|context| {
+            loop {
+                cancellation.check()?;
+                self.check_monitor(state)?;
+                state.check_interest(Interest::Readable)?;
+                if wait_first && !registration.is_ready() {
+                    let controller = route.get_or_insert_with(|| self.route_ref());
+                    if cancelled.as_mut().poll(context).is_ready() {
+                        return Poll::Ready(Err(core_scheduler::Cancelled.into()));
+                    }
+                    std::task::ready!(
+                        self.poll_monitor_wait(monitor, controller, &mut wait, context)
+                    )?;
+                }
+                let readiness = registration.readiness.load(Ordering::Acquire);
+                match syscall("monitor-recv", || {
+                    crate::retained_socket::recv(state.socket(), buffer, SocketRecvFlags::default())
+                }) {
+                    Ok(count) => {
+                        if count > 0 {
+                            *established = true;
+                            if count < buffer.len() {
+                                registration.clear_readiness(readiness);
+                            }
+                        }
+                        return Poll::Ready(Ok(count));
+                    }
+                    Err(error) if error.kind() == io::ErrorKind::WouldBlock => {
+                        registration.clear_readiness(readiness);
+                        wait_first = true;
+                    }
+                    Err(error) if error.kind() == io::ErrorKind::Interrupted => {}
+                    Err(error) => return Poll::Ready(Err(error.into())),
+                }
+            }
+        })
+        .await
+    }
